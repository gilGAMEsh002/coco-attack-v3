cd coco_attack

  --repo-dir .. --assets-dir ../cocota_data_eval_result \

.venv/bin/coco-attack materialize-prompts \
  --repo-dir .. --assets-dir ../cocota_data_eval_result --data-dir /tmp/prepared \
  --combination cwe078-0 --oracle-id cwe078-0 --form all --output-dir /tmp/prompts

.venv/bin/coco-attack clean-generations \
  --data-dir /tmp/prepared \
  --input-jsonl ../cocota_data_eval_result/runs_by_model_old/gpt-4o/rep1/cwe078_cocota_gpt-4o_t0p7_r5/outputs/generations.jsonl \
  --combination cwe078-0 --oracle-id cwe078-0 --output-dir /tmp/cleaned

.venv/bin/coco-attack evaluate-static \
  --assets-dir ../cocota_data_eval_result --data-dir /tmp/prepared --cleaned-dir /tmp/cleaned \
  --combination cwe078-0 --oracle-id cwe078-0 --config "path/to/eval_config.json" --output-dir /tmp/evaluated

.venv/bin/coco-attack calibrate-references \
  --assets-dir ../cocota_data_eval_result --data-dir /tmp/prepared \
  --config "path/to/calib_config.json" --output-dir /tmp/calibrated

.venv/bin/coco-attack compare-history \
  --assets-dir ../cocota_data_eval_result --data-dir /tmp/prepared \
